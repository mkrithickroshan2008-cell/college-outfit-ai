import OpenAI from "openai"; import formidable from "formidable"; import fs from "fs"; export const config = { api: { bodyParser: false } }; export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed" }); } try { const form = formidable({ multiples: false }); const [fields, files] = await form.parse(req); const photo = files.photo?.[0]; if (!photo) { return res.status(400).json({ error: "Photo is required" }); } const imageBuffer = fs.readFileSync(photo.filepath); const base64Image = imageBuffer.toString("base64"); const mimeType = photo.mimetype || "image/jpeg"; const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY }); const response = await client.responses.create({ model: "gpt-4.1-mini", input: [ { role: "user", content: [ { type: "input_text", text: ` Analyze this person's outfit and give a practical college outfit recommendation. Return: 1. Current outfit 2. What looks good 3. Recommended shirt 4. Recommended pants 5. Recommended shoes 6. Best colour combination 7. Accessories 8. Overall style rating out of 10 Keep the recommendation realistic, simple and suitable for college. Do not identify the person's identity. ` }, { type: "input_image", image_url: `data:${mimeType};base64,${base64Image}` } ] } ] }); return res.status(200).json({ recommendation: response.output_text }); } catch (error) { console.error(error); return res.status(500).json({ error: "AI analysis failed. Check the backend configuration." }); } }
